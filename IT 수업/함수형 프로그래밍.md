익명 클래스와 람다식 관련 예시

```java

package chapter13;

import java.util.Arrays;
import java.util.List;
import java.util.Random;
import java.util.stream.Collectors;

public class LambdaTestT {

    static int check(int[] data, NumFilterTest filter) {
        int num = 0;
        for (int d : data) {
            if (filter.numCheck(d)) {
                num++;
            }
        }
        return num;
    }

    public static void main(String[] args) {

        int[] nums = {10, 25, 35, 9, 40, 20, 35, 11, 77, 55};

        // 익명클래스 사용
        //                      이부분이 class ??? implements NumFilterTest { } 라는 익명클래스로 컴파일 된다.
        NumFilterTest filter1 = new NumFilterTest() {
            @Override
            public boolean numCheck(int a) {
                return a > 30;
            }
        };
        System.out.println(check(nums, filter1));

        // 추상메서드가 한개인 경우에만 가능한 방법이며, 
        // 우변의 로직이 numCheck 의 로직으로 자동으로 추론 됨
        NumFilterTest filter2 = num -> num > 30;
        System.out.println(check(nums, filter2));


        // 이렇게 바로 쓸 수도 있음! check 메서드의 매개변수가 NumFilter 인터페이스기 때문에! ( 함수형 인터페이스 )
        System.out.println(check(nums, num -> num > 30));
    }
}

// 함수형 인터페이스라는 선언, 이걸 붙이는 순간 추상 메서드는 하나만 선언 가능해짐
// @FunctionalInterface 어노테이션의 역할은, 추상메서드를 두개 이상 선언하려고 하면 에러를 발생 시켜 줌
@FunctionalInterface
interface NumFilterTest {

    // 이렇게 추상메서드가 한개이기만 하면 된다.
    boolean numCheck(int a);

    /*
    이런식으로 인터페이스에서 구현이 가능한 default 메서드나, static 메서드는 몇개라도 선언 가능함.
    default int tetsMethod(int num1) {
        return num1;
    }*/
}

```



오늘 배운 자바에서 지원하는 함수형 인터페이스들 정리

( Predicate , Function , BiFunction )



1. Predicate<T> : **test 라는 추상메서드**를 지원하며, **하나의 T타입 변수를 입력** 받아, 내가 선언한 로직에 해당 되면 **True , 아니면 False 를 반환**해 준다. 
   
   ( 입력 받는 타입은 내가 선언하기 나름 )
   
   ```java
   package chapter13;
   
   import java.util.function.Predicate;
   
   public class LambdaTest2 {
   
       public static void main(String[] args) {
   
           // 이렇게 람다식으로 선언한 로직에 결과를 true / false로 반환해주는게 .test() 의 역할!
           // Predicate<T> 에서 입력받을 타입인 T 나, test 메서드의 로직은 아래처럼 선언하는 시점에 지정을 해준다
           Predicate<Integer> isEven = n -> n%2 == 0;
           Predicate<Integer> isMoreThanThirty = n -> n > 30;
   
           System.out.println(isEven.test(30));    // true
           System.out.println(isEven.test(30));    // false
   
           // 이렇게 여러 로직들을 .or 또는 .and로 이을 수 있다
           // 이건 짝수이면서 && 30 초과의 숫자냐를 물어보는 것
           System.out.println(isEven.and(isMoreThanThirty).test(38)); // true
           System.out.println(isEven.and(isMoreThanThirty).test(37)); // false
       }
   
   }
   
   
   ```



2. Function<T, R> : **apply라는 추상 메서드**를 지원하며, 
   
   **하나의 변수 (T 타입 ) 를 입력** 받아 내가 정한 로직을 거친 후 **R 타입으로 반환**해 줌 ( 입력받는 타입인 T,  반환 타입인 R 은 내가 정하기 나름)

```java
package chapter13;

import java.util.Random;
import java.util.function.Function;

public class LambdaFunctionTest {
    
                                        // 이렇게 매개변수를 함수형 인터페이스 ( 혹은 함수형 클래스) 로 설정하면
                                        // 함수 호출시 이 자리에 람다식을 사용 할 수 있음
    static void multiplier(int[] data, Function<Integer, Integer> func) {
        for (int i = 0; i < data.length; i++) {
            int r = func.apply(data[i]);
            data[i] = r;
        }
    }

    public static void main(String[] args) {
        Random random = new Random();

        int[] nums = new int [10];

        for (int i = 0; i < 10; i++) {
            nums[i] = random.nextInt(11, 31);
        }

        for (int num : nums) {
            System.out.printf("%d ", num);
        }
        System.out.println();
        
        Function<Integer, Integer> setTriple = num -> 3*num;
        
        multiplier(nums, setTriple);
        // multiplier의 두번째 인자가 함수형 인터페이스이므로,
        // multiplier(nums, num -> 3*num) 으로도 선언 가능

        for (int num : nums) {
            System.out.printf("%d ", num);
        }
        System.out.println();


        // Function<T,R> 에서  T 와 R 은 어떤 타입이던 내가 정하기 나름이다.
        // String 타입을 입력 받아, Integer 로 반환해주며, 로직은 입력받은 문자열의 길이를 반환
        // apply 의 로직도 내가 작성하기 나름 
        // ( 다만 함수형 프로그래밍의 특성상 람다식에 복잡한 로직을 입력하는건 지양해야 한다. ) 
        Function<String, Integer> calcStringLength = str -> str.length();

        System.out.println(calcStringLength.apply("Hello Java!"));  // 11
    }
}

```



3. BiFunction<T, U, R> : Function 과 유사하게 **apply 라는 메서드**를 지원하며
   
   **두 개의 변수 ( T타입,  U 타입) 을 입력**받아 내가 설정한 로직을 거쳐  **R 타입으로 반환**해 준다.

```java

package chapter13;

import java.util.function.BiFunction;

public class LambdaTest4 {
    public static void main(String[] args) {
        
        // BiFunction<T, U, R> 
        // 얘도 apply 메서드이며, T , U 두개 인자를 입력 받아 이를 이용한 특정 로직의 결과를 R 로 반환해줌
        // Function 은 인자를 하나만 입력 받을 수 있으나, BiFunction 은 인자 두개를 입력 받아 처리 가능하다는 차이가 있음
        BiFunction<Integer, Double, Double> add = (a, b) -> {
            System.out.println("num1 = " + a + " num2 = " + b);
            return a+b;
        };
        System.out.println(add.apply(10, 20.5));

        BiFunction<String, Integer, Boolean> chkStrLength = (str, num) -> str.length() > num;
        System.out.println(chkStrLength.apply("바람의 나라", 10));       // false
        System.out.println(chkStrLength.apply("거센바람 높은파도가 우리앞길 막아서도", 10)); // true
        

    }
}

```
